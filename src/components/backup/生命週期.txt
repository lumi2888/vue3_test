<!--Practice.vue 子組件-->
<template>
    <div class="practice">
      <h2>當前求和為:{{sum}}</h2>
      <button @click="add">sum</button>
    </div>
</template>

<script lang="ts" setup name="Practice">
import {ref,onBeforeMount,onMounted,onBeforeUpdate,onUpdated,onBeforeUnmount,onUnmounted} from 'vue'

let sum = ref(0)

function add(){
   sum.value += 1
}
// create
console.log('create')

// mount
onBeforeMount(()=>{
   console.log('mount before')
})
// mount complete(完成)
onMounted(()=>{
   console.log('子--mount complete')
})

// update before
onBeforeUpdate(()=>{
   console.log('update before')
})
// update complete
onUpdated(()=>{
   console.log('update complete')
})

//unmount before
onBeforeUnmount(()=>{
   console.log('unmount before')
})
//unmount complete
onUnmounted(()=>{
   console.log('unmount complete')
})
